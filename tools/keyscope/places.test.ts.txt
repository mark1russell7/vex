// Throwaway measure: the places of the program of each module of ireko, and of the corpus.
import { domainsOf, programOf } from "@mark1russell7/vex-mcp";
import type { Expr } from "@mark1russell7/vex";
import { it } from "vitest";
import { placesOf } from "../../compiler/src/unshared.ts";
import { BETWEEN_PROGRAM, BOUNDS_PROGRAM, OVERLAP_PROGRAM } from "./ireko-audit.ts";
import { readCorpora } from "./corpus.ts";
import { GRID_PROGRAM } from "./ireko-grid.ts";
import { LAYERS_PROGRAM } from "./ireko-layers.ts";
import { MEASURE_COLUMNS } from "./ireko-measure.ts";
import { PLACE_COLUMNS } from "./ireko-place.ts";
import { RADIAL_PROGRAM } from "./ireko-radial.ts";
import { AVOID_PROGRAM, EDGE_DIRECTION_PROGRAM, LABEL_BOX_PROGRAM, OBSTACLES_PROGRAM, ORTHOGONAL_PROGRAM, STRAIGHT_PROGRAM } from "./ireko-route.ts";
import { NODE_SHAPE } from "./ireko-shapes.ts";
import { TREE_PROGRAM } from "./ireko-tree.ts";

it("places", () => {
  const rows: Readonly<Record<string, readonly Expr[]>> = {
    grid: [GRID_PROGRAM], radial: [RADIAL_PROGRAM], layers: [LAYERS_PROGRAM], tree: [TREE_PROGRAM], shape: [NODE_SHAPE],
    place: [...PLACE_COLUMNS.values()], measure: [...MEASURE_COLUMNS.values()],
    straight: [STRAIGHT_PROGRAM], orthogonal: [ORTHOGONAL_PROGRAM], avoid: [AVOID_PROGRAM], direction: [EDGE_DIRECTION_PROGRAM], label: [LABEL_BOX_PROGRAM], obstacles: [OBSTACLES_PROGRAM],
    overlap: [OVERLAP_PROGRAM], bounds: [BOUNDS_PROGRAM], between: [BETWEEN_PROGRAM],
  };
  for (const [name, exprs] of Object.entries(rows)) console.log(`PLACES ${name} ${String(placesOf(exprs))}`);
  let most = 0;
  for (const corpus of readCorpora()) for (const c of corpus.programs) {
    const domains = domainsOf(c.domains);
    most = Math.max(most, placesOf([programOf(c.program, domains) as Expr]));
  }
  console.log(`PLACES corpus most ${String(most)}`);
}, 600_000);
